option(SHIN_FETCH_SDL "Build pinned SDL dependencies from source" OFF)
if(SHIN_FETCH_SDL)
 enable_language(C)
 include(FetchContent)
 set(SDL_SHARED OFF CACHE BOOL "" FORCE)
 set(SDL_STATIC ON CACHE BOOL "" FORCE)
 set(SDL_TEST OFF CACHE BOOL "" FORCE)
 set(SDL2_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
 set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
 foreach(codec FLAC MOD MP3 MIDI OPUS VORBIS WAVPACK GME CMD)
  set(SDL2MIXER_${codec} OFF CACHE BOOL "" FORCE)
 endforeach()
 set(SDL2MIXER_SAMPLES OFF CACHE BOOL "" FORCE)
 set(SDL2MIXER_INSTALL OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(SDL2 URL https://github.com/libsdl-org/SDL/releases/download/release-2.30.11/SDL2-2.30.11.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
 FetchContent_MakeAvailable(SDL2)
 FetchContent_Declare(SDL2_mixer URL https://github.com/libsdl-org/SDL_mixer/releases/download/release-2.8.1/SDL2_mixer-2.8.1.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
 FetchContent_MakeAvailable(SDL2_mixer)
 set(SHIN_SDL_TARGET SDL2::SDL2-static)
 set(SHIN_MIXER_TARGET SDL2_mixer::SDL2_mixer-static)
else()
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
 pkg_check_modules(SDL2MIXER REQUIRED IMPORTED_TARGET SDL2_mixer)
 set(SHIN_SDL_TARGET PkgConfig::SDL2)
 set(SHIN_MIXER_TARGET PkgConfig::SDL2MIXER)
endif()
add_executable(shin_portable MACOSX_BUNDLE main.cpp)
target_link_libraries(shin_portable PRIVATE shin_game ${SHIN_SDL_TARGET} ${SHIN_MIXER_TARGET})
set_target_properties(shin_portable PROPERTIES OUTPUT_NAME SHIN MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in" MACOSX_BUNDLE_BUNDLE_NAME "蜡笔小新1-重构版" MACOSX_BUNDLE_GUI_IDENTIFIER "org.libaichuan.shin-rebuild" MACOSX_BUNDLE_SHORT_VERSION_STRING "0.3.26" MACOSX_BUNDLE_BUNDLE_VERSION "0.3.26")
install(TARGETS shin_portable BUNDLE DESTINATION . RUNTIME DESTINATION .)
